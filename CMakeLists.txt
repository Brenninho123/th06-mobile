    set(TH06_VENDORED_SDL_DEFAULT ON)
else()
    set(TH06_VENDORED_SDL_DEFAULT OFF)

if(TH06_VENDORED_SDL)
    foreach(dep SDL SDL_image SDL_ttf)
        if(NOT EXISTS "${TH06_DEPS_DIR}/${dep}/CMakeLists.txt")
            message(FATAL_ERROR "Missing ${dep}. Run: git submodule update --init --recursive")
        endif()
    endforeach()

    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
    add_subdirectory("${TH06_DEPS_DIR}/SDL" "${CMAKE_CURRENT_BINARY_DIR}/deps/SDL" EXCLUDE_FROM_ALL)

    if(NOT TARGET SDL2::SDL2-static)
        add_library(SDL2::SDL2-static ALIAS SDL2)
    endif()

    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)

    set(SDL2IMAGE_VENDORED OFF CACHE BOOL "" FORCE)
    set(SDL2IMAGE_DEPS_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL2IMAGE_BACKEND_STB ON CACHE BOOL "" FORCE)
    set(SDL2IMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
    set(SDL2IMAGE_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL2IMAGE_INSTALL OFF CACHE BOOL "" FORCE)
    foreach(fmt AVIF JXL TIF WEBP SVG XCF XPM XV LBM PCX PNM QOI TGA GIF)
        set(SDL2IMAGE_${fmt} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory("${TH06_DEPS_DIR}/SDL_image" "${CMAKE_CURRENT_BINARY_DIR}/deps/SDL_image" EXCLUDE_FROM_ALL)

    set(SDL2TTF_VENDORED ON CACHE BOOL "" FORCE)
    set(SDL2TTF_HARFBUZZ OFF CACHE BOOL "" FORCE)
    set(SDL2TTF_SAMPLES OFF CACHE BOOL "" FORCE)
    set(SDL2TTF_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${TH06_DEPS_DIR}/SDL_ttf" "${CMAKE_CURRENT_BINARY_DIR}/deps/SDL_ttf" EXCLUDE_FROM_ALL)

    file(MAKE_DIRECTORY "${TH06_COMPAT_INCLUDE_DIR}/SDL2")
    configure_file("${TH06_DEPS_DIR}/SDL_image/include/SDL_image.h" "${TH06_COMPAT_INCLUDE_DIR}/SDL2/SDL_image.h" COPYONLY)
    configure_file("${TH06_DEPS_DIR}/SDL_ttf/SDL_ttf.h" "${TH06_COMPAT_INCLUDE_DIR}/SDL2/SDL_ttf.h" COPYONLY)

    set(TH06_SDL_LIBS SDL2::SDL2 SDL2_image SDL2_ttf)
else()
    find_package(SDL2 REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2_EXTRA REQUIRED IMPORTED_TARGET SDL2_image SDL2_ttf)
    set(TH06_SDL_LIBS SDL2::SDL2 PkgConfig::SDL2_EXTRA)
endif()

function(th06_embed_file input variable output)
    file(READ "${input}" contents HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1, " contents "${contents}")
    file(APPEND "${output}" "\nstatic char ${variable}[] = {${contents}0};\n")
endfunction()

set(TH06_EMBEDS_HEADER "${TH06_GENERATED_DIR}/autogenerated/Embeds.hpp")
file(MAKE_DIRECTORY "${TH06_GENERATED_DIR}/autogenerated")
file(WRITE "${TH06_EMBEDS_HEADER}" "#pragma once\n")
th06_embed_file("${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.vert" vertShaderBytes "${TH06_EMBEDS_HEADER}")
th06_embed_file("${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.frag" fragShaderBytes "${TH06_EMBEDS_HEADER}")

    "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.vert"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.frag")

    target_include_directories(${TH06_TARGET} PRIVATE "${TH06_COMPAT_INCLUDE_DIR}")

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(${TH06_TARGET} PRIVATE -Wall -Wextra -Wno-gnu-anonymous-struct -Wno-inconsistent-missing-override)
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(${TH06_TARGET} PRIVATE -Wall -Wextra -fjump-tables)
endif()
